23static inline int eat_cr(unsigned char **p, unsigned char *end) {
24 // On vérifie si on est hors limites
25 if (*p >= end)
26 return 0;
27 // Si on a un CR
28 if (is_CR(**p)) {
29 // On mange le CR
30 (*p)++;
31 if (*p >= end) {
32 return 1;
33 }
34 // Si on a un LF juste après (CRLF)
35 if (is_EOL(**p)) {
36 // On mange le LF
37 (*p)++;
38 }
39 // On a mangé un CR ou un CRLF
40 return 1;
41 }
42
43 // Si on a un LF seul
44 if (is_EOL(**p)) {
45 // On mange le LF
46 (*p)++;
47 return 1;
48 }
49
50 // Dans tous les autres cas, on ne mange rien
51 return 0;
52}

55PS *psopen(int fd, size_t s, int M) {
56 unsigned char *base = mmap(NULL, s, PROT_READ, MAP_PRIVATE, fd, 0);
57 if (base == MAP_FAILED) {
58 close(fd);
59 return NULL;
60 }
61
62 // On alloue notre zone mémoire pour le parser
63 PS *ps = (PS *)calloc(1, sizeof(PS));
64 if (!ps) {
65 if (base) {
66 munmap((void *)base, s);
67 }
68 close(fd);
69 return NULL;
70 }
71
72 ps->fd = fd; // File descriptor
73 ps->base = base; // Pointeur vers le début du mmap
74 ps->cur = base; // Pointeur courant
75 ps->end = base + s; // Fin de la zone mappée
76 ps->size = s; // Espace mémoire
77 ps->M = M; // Mu (μ dans l'énoncé)
78 ps->para_s = ps->cur; // Début du paragraphe
79 ps->para_e = ps->cur; // Fin du paragraphe
80
81 return ps;
82}

85int nextPG(PS *ps) {
86
87 /* La fin d'un paragraphe est ainsi définie : (\r\n|\r|\n) [ \t|' ']* (\r\n|\r|\n), càd, un saut de ligne, zéro ou plusieurs espaces, et un autre saut de ligne. */
88 if (!ps) {
89 return -1;
90 }
91 unsigned char *p, *end;
92 p = ps->cur;
93 end = ps->end;
94
95 /* ON SUPPRIME LES ESPACES EN DÉBUT (ET EN FIN, cf. return ci-dessous) DE PARAGRAPHE */
96
97 while (p < end) {
98 // On saute directement les espaces
99 if (is_SPACE(*p)) {
100 p++;
101 // Pour les CR/LF/CRLF on utilise la fonction dédiée
102 } else if (is_CRLF(*p)) {
103 eat_cr(&p, end);
104 } else {
105 // Début d'un mot
106 break;
107 }
108 }
109
110 if (p >= end) {
111 // On met à jour la position courante
112 ps->cur = p;
113 return 0;
114 }
115
116 // On marque le "vrai" début du paragraphe
117 unsigned char *start = p;
118
119 /* ON CHERCHE LA FIN (DEUX CR|LF CONSÉCUTIFS) DU PARAGRAPHE */
120
121 // Booléen qui indique la présence d'un CR
122 int crlf = 0;
123
124 // On scanne le texte
125 while (p < end) {
126 // On n'a encore aucun saut de ligne
127 if (!crlf) {
128 // Quand on en trouve un on le mange et on active le booléen
129 if (is_CRLF(*p)) {
130 eat_cr(&p, end);
131 crlf = 1;
132 continue;
133 }
134 // Pas de CR, on continue
135 p++;
136 } else {
137 // on a vu un CR, tant qu'on a des espaces on peut les ignorer
138 while (p < end && is_SPACE(*p)) {
139 p++;
140 }
141 // Si on est arrivé à la fin du fichier
142 if (p >= end) {
143 // Par souci de cohérence
144 crlf = 0;
145 break;
146 }
147 // si le prochain caractère après 0 ou n espaces est un CR on a trouvé une fin de paragraphe
148 if (is_CRLF(*p)) {
149 eat_cr(&p, end);
150 break;
151 } else {
152 // Sinon c'était simplement un retour à la ligne dans un paragraphe, on reset le booléen
153 crlf = 0;
154 }
155 }
156 }
157
158 ps->para_s = start;
159 ps->para_e = p;
160
161 // On se positionne après le paragraphe pour la prochaine lecture
162 ps->cur = p;
163 return 1;
164}

167int getPG(PS *ps, PG *out, int *p_err) {
168 if (!ps || !out)
169 return 0;
170
171 unsigned char *s = ps->para_s;
172 unsigned char *e = ps->para_e;
173 size_t len = (size_t)(e - s);
174
175 // Maximum de mots théorique : (taille du paragraphe + 1) / 2 (1 caractère + 1 séparateur)
176 size_t max = (len + 1) / 2 + 1;
177
178 uint16_t *W = (uint16_t *)malloc(max * sizeof(*W));
179 unsigned char **P = (unsigned char **)malloc(max * sizeof(*P));
180 if (!P || !W) {
181 free(P);
182 free(W);
183 return 0;
184 }
185
186 long long n = 0;
187 int in_word = 0;
188 unsigned char *wstart = NULL;
189 long long l = 0;
190
191 unsigned char *p = s;
192 while (p < e) {
193 unsigned char c = *p;
194
195 // Fin éventuelle de mot
196 if (is_SEP(c)) {
197
198 // On est déjà dans un mot
199 if (in_word) {
200
201 // On a dépassé la longueur maximale autorisée
202 if (l > (long long)ps->M) {
203 if (p_err) {
204 *p_err = 1;
205 }
206 free(P);
207 free(W);
208 return 0;
209 }
210 P[n] = wstart;
211 W[n] = (uint16_t)l;
212 n++;
213 in_word = 0;
214 l = 0;
215 }
216 p++;
217 continue;
218 }
219
220 // On n'est pas dans un mot
221 if (!in_word) {
222 in_word = 1;
223 wstart = p;
224 l = 1;
225 } else {
226 l++;
227 }
228 p++;
229 }
230
231 // mot final éventuel
232 if (in_word) {
233 if (l > (long long)ps->M) {
234 if (p_err) {
235 *p_err = 1;
236 }
237 free(P);
238 free(W);
239 return 0;
240 }
241 P[n] = wstart;
242 W[n] = (uint16_t)l;
243 n++;
244 }
245
246 out->n = n;
247 out->base = ps->base;
248 out->P = P;
249 out->W = W;
250 return 1;
251}
